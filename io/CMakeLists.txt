add_subdirectory(serial)
add_library(io STATIC camera.cpp cboard.cpp hikrobot/hikrobot.cpp)
target_include_directories(io PUBLIC hikrobot/include)
set(HIKROBOT_SDK_LIB_DIR "" CACHE PATH "Directory containing the complete Hikrobot SDK libraries")
if(NOT HIKROBOT_SDK_LIB_DIR AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64" AND EXISTS "/opt/MVS/lib/64/libMvCameraControl.so")
  set(HIKROBOT_SDK_LIB_DIR "/opt/MVS/lib/64" CACHE PATH "Directory containing the complete Hikrobot SDK libraries" FORCE)
endif()
if(HIKROBOT_SDK_LIB_DIR)
  if(NOT EXISTS "${HIKROBOT_SDK_LIB_DIR}/libMvCameraControl.so")
    message(FATAL_ERROR "HIKROBOT_SDK_LIB_DIR must contain libMvCameraControl.so")
  endif()
  target_link_directories(io PUBLIC ${HIKROBOT_SDK_LIB_DIR})
  message(STATUS "Hikrobot SDK libraries: ${HIKROBOT_SDK_LIB_DIR}")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64")
  target_link_directories(io PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/hikrobot/lib/amd64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  target_link_directories(io PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/hikrobot/lib/arm64)
else()
  message(FATAL_ERROR "Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
target_link_libraries(io PUBLIC MvCameraControl usb-1.0 yaml-cpp serial tools Threads::Threads)
